\begin{lemma}\label{lem:sup_quadratic}
For all $u \in (0,1)$,
\begin{align*}
  \sup_{\lambda \ge 0} \frac{1}{2}\log (1 + \lambda (u - 1)) + \frac{1}{2}\log (1 + \lambda (u + 1))
  = \frac{1}{2} \log \frac{1}{1 - u^2}
  \: .
\end{align*}
\end{lemma}

\begin{proof}
It suffices to compute the supremum of $(1 + \lambda (u - 1))(1 + \lambda (u + 1))$ over $\lambda \ge 0$, which is simply finding the maximum of a quadratic function.
\end{proof}


\begin{lemma}\label{lem:inf_symm_sup_utility}
For all $u \in (0,1)$,
\begin{align*}
  \inf_{R \in \mathcal{P}([-1,1]), \ -\id \circ R = R} \sup_{\lambda \ge 0} R[\log (1 + \lambda (X + u)]]
  = \frac{1}{2} \log \frac{1}{1 - u^2}
  \: .
\end{align*}
\end{lemma}

\begin{proof}\uses{lem:sup_quadratic}
For the first inequality, we use that the infimum is less than the value at $R = \frac{1}{2}\delta_{-1} + \frac{1}{2}\delta_{1}$.
Then by Lemma~\ref{lem:sup_quadratic}, the supremum over $\lambda \ge 0$ for that distribution is $\frac{1}{2} \log \frac{1}{1 - u^2}$.

For the second inequality, we use first the inequality $\inf\sup \ge \sup\inf$.
\begin{align*}
  &\inf_{R \in \mathcal{P}([-1,1]), \ -\id \circ R = R} \sup_{\lambda \ge 0} R[\log (1 + \lambda (X + u))]
  \\
  &\ge \sup_{\lambda \ge 0} \inf_{R \in \mathcal{P}([-1,1]), \ -\id \circ R = R} R[\log (1 + \lambda (X + u))]
  \\
  &= \sup_{\lambda \ge 0} \inf_{R \in \mathcal{P}([0,1])} \frac{1}{2} R[\log (1 + \lambda (X + u))] + \frac{1}{2}R[\log(1 + \lambda (-X + u))]
  \: .
\end{align*}
For any $\lambda \ge 0$, the infimum is achieved by a Dirac distribution $\delta_{x}$ for some $x \in [0,1]$.
We get a lower bound
\begin{align*}
  \sup_{\lambda \ge 0} \inf_{x \in [0,1]} \frac{1}{2} \log (1 + \lambda (u + x)) + \frac{1}{2} \log (1 + \lambda (u - x))
  \: .
\end{align*}
The minimizer for $x$ is at $x=1$ for all values of $\lambda$.
We get to the value
\begin{align*}
  \sup_{\lambda \ge 0} \frac{1}{2} \log (1 + \lambda (u + 1)) + \frac{1}{2} \log(1 + \lambda (u - 1))
\end{align*}
which is equal to $\frac{1}{2} \log \frac{1}{1 - u^2}$ by Lemma~\ref{lem:sup_quadratic}.
\end{proof}

\begin{lemma}\label{lem:utility_fin_gen_ge}
  \uses{def:eVar}
Let $\Phi = \{\phi_1, \ldots, \phi_d\}$ be a set of measurable functions from $\mathcal{X}$ to $\mathbb{R}$.
Let $\mathcal{P}^\Phi = \{P \in \mathcal{P}(\mathcal{X}) \mid \forall \phi \in \Phi, \ P[\vert \phi \vert] < \infty \text{ and } P[\phi] \le 0\}$.
Denote by $\phi : \mathcal{X} \to \mathbb{R}^d$ the vector-valued function $\phi(x) = (\phi_1(x), \ldots, \phi_d(x))$.
Then
\begin{align*}
  \sup_{f \in \mathcal{E}_{\mathcal{P}^\Phi}}R[\log f]
  \ge \sup_{f \in \mathcal{E}_{\{P \in \mathcal{P}(\mathbb{R}^d) \mid \forall i, P[X_i] \le 0\}}}(\phi \circ R)[\log f]
  \; .
\end{align*}
\end{lemma}

\begin{proof}
  \uses{thm:dataprocessing, lem:eVarSet_anti}
By Theorem~\ref{thm:dataprocessing} for the function $\phi$, we have
\begin{align*}
  \sup_{f \in \mathcal{E}_{\mathcal{P}^{\Phi}}} R[\log f]
  &\ge \sup_{f \in \mathcal{E}_{\phi \circ \mathcal{P}^\Phi}} (\phi \circ R)[\log f]
  \: .
\end{align*}
$\phi \circ \mathcal{P}^\Phi \subseteq \{P \in \mathcal{P}(\mathbb{R}^d) \mid \forall i, P[X_i] \le 0\}$ and their e-variable sets are in the reversed order (Lemma~\ref{lem:eVarSet_anti}).
We get the result of the lemma.
\end{proof}


\begin{lemma}\label{lem:todo2}
Let $\mathcal{P},\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ be two sets of probability measures.
Suppose that there exists a measurable function $f_0 : \mathcal{X} \to [-1, 1]$ and $\varepsilon \in (0,1]$ such that
\begin{align*}
  \forall P \in \mathcal{P}, \: P[f_0] &\le - \varepsilon
  \: , \\
  \forall Q \in \mathcal{Q}, \: Q[f_0] &\ge \varepsilon
  \: .
\end{align*}
Then
\begin{align*}
  \inf_{R \in \mathcal{P}(\mathcal{X})}\left\{\sup_{f \in \mathcal{E}_{\mathcal{P}}} R[\log f]
    + \sup_{g \in \mathcal{E}_{\mathcal{Q}}} R[\log g] \right\}
  \ge \inf_{R \in \mathcal{P}([-1,1])}\left\{\sup_{f \in \mathcal{E}_{\{P \mid P[\id] \le -\varepsilon\}}} R[\log f]
  + \sup_{g \in \mathcal{E}_{\{Q \mid Q[\id] \ge \varepsilon\}}} R[\log g] \right\}
  \: .
\end{align*}
\end{lemma}

\begin{proof}
  \uses{thm:dataprocessing, lem:eVarSet_anti}
By Theorem~\ref{thm:dataprocessing} for the function $f_0$, we have
\begin{align*}
  \inf_{R \in \mathcal{P}(\mathcal{X})}\left\{\sup_{f \in \mathcal{E}_{\mathcal{P}}} R[\log f]
    + \sup_{g \in \mathcal{E}_{\mathcal{Q}}} R[\log g] \right\}
  &\ge \inf_{R \in \mathcal{P}(\mathcal{X})}\left\{\sup_{f \in \mathcal{E}_{f_0 \circ \mathcal{P}}} (f_0 \circ R)[\log f]
    + \sup_{g \in \mathcal{E}_{f_0 \circ \mathcal{Q}}} (f_0 \circ R)[\log g] \right\}
  \\
  &\ge \inf_{R \in \mathcal{P}([-1,1])}\left\{\sup_{f \in \mathcal{E}_{f_0 \circ \mathcal{P}}} R[\log f]
    + \sup_{g \in \mathcal{E}_{f_0 \circ \mathcal{Q}}} R[\log g] \right\}
  \: .
\end{align*}
$f_0 \circ \mathcal{P} \subseteq \{P \in \mathcal{P}([0,1]) \mid P[\id] \le -\varepsilon\}$ and their e-variable sets are in the reversed order (Lemma~\ref{lem:eVarSet_anti}).
A similar inclusion holds for $f_0 \circ \mathcal{Q}$.
We get the result of the lemma.
\end{proof}

\begin{theorem}\label{thm:separation_lower_bound}
  \uses{def:eVar}
Let $\mathcal{P},\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ be two sets of probability measures.
Suppose that there exists a measurable function $f_0 : \mathcal{X} \to [0, 1]$ and $\delta \in (0,1/2)$ such that
\begin{align*}
  \forall P \in \mathcal{P}, \: P[f_0] &\le \delta
  \: , \\
  \forall Q \in \mathcal{Q}, \: Q[f_0] &\ge 1 - \delta
  \: .
\end{align*}
Then
\begin{align*}
  \inf_{R \in \mathcal{P}(\mathcal{X})}\left\{\sup_{f \in \mathcal{E}_{\mathcal{P}}} R[\log f]
    + \sup_{g \in \mathcal{E}_{\mathcal{Q}}} R[\log g] \right\}
  &\ge \log \frac{1}{4\delta(1 - \delta)}
  \: .
\end{align*}
\end{theorem}

This is analogous to a lower bound on the Rényi divergence with parameter $\alpha = 1/2$.

\begin{proof}
  \uses{lem:todo}
The function $\tilde{f}_0 : \mathcal{X} \to [-1, 1]$ defined by $\tilde{f}_0(x) = 2f_0(x) - 1$ satisfies the assumptions of Lemma~\ref{lem:todo} with $\varepsilon = 1 - 2 \delta$.
We get
\begin{align*}
  \inf_{R \in \mathcal{P}(\mathcal{X})}\left\{\sup_{f \in \mathcal{E}_{\mathcal{P}}} R[\log f]
    + \sup_{g \in \mathcal{E}_{\mathcal{Q}}} R[\log g] \right\}
  &\ge \log \frac{1}{1 - (1 - 2\delta)^2}
  = \log \frac{1}{4\delta(1 - \delta)}
  \: .
\end{align*}
\end{proof}

\section{Data processing inequality through KL duality}

\begin{align*}
  \sup_{f \in \mathcal{E}_{\mathcal{Q}}} P[\log f]
  &= \inf_{Q \in \mathcal{Q}_{\eff}} \KL(P, Q)
  \\
  &\ge \inf_{Q \in \mathcal{Q}_{\eff}} \KL(\kappa \circ P, \kappa \circ Q)
  \\
  &= \inf_{Q' \in \kappa \circ \mathcal{Q}_{\eff}} \KL(\kappa \circ P, Q')
  \\
  &\ge \inf_{Q' \in (\kappa \circ \mathcal{Q})_{\eff}} \KL(\kappa \circ P, Q')
  \\
  &= \sup_{g \in \mathcal{E}_{\kappa \circ \mathcal{Q}}} (\kappa \circ P)[\log g]
  \: .
\end{align*}

The first inequality is the data processing inequality for KL divergence.
The second inequality comes from $\kappa \circ \mathcal{Q}_{\eff} \subseteq (\kappa \circ \mathcal{Q})_{\eff}$, which we proved in Lemma~\ref{lem:effectiveSet_comp_subset}.
