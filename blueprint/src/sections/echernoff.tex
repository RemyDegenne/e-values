\chapter{e-Rényi and e-Chernoff divergences}

\begin{definition}\label{def:erenyiDiv}
  \leanok
  \lean{ProbabilityTheory.erenyiDiv}
\begin{align*}
  \eR_\alpha(\mathcal{P}, \mathcal{Q})
  &= \frac{1}{1 - \alpha} \inf_{R \in \mathcal{P}(\mathcal{X})} \left( \alpha \sup_{f \in \mathcal{E}_{\mathcal{P}}} R[\log f] + (1 - \alpha) \sup_{g \in \mathcal{E}_{\mathcal{Q}}} R[\log g] \right)
  \: .
\end{align*}
\end{definition}


\begin{definition}\label{def:echernoffDiv}
  \leanok
  \lean{ProbabilityTheory.echernoffDiv}
\begin{align*}
  \eC_\alpha(\mathcal{P}, \mathcal{Q})
  &= \inf_{R \in \mathcal{P}(\mathcal{X})} \max\left\{\sup_{f \in \mathcal{E}_{\mathcal{P}}} R[\log f], \sup_{g \in \mathcal{E}_{\mathcal{Q}}} R[\log g] \right\}
  \: .
\end{align*}
\end{definition}


\begin{lemma}\label{lem:erenyiDiv_anti}
  \uses{def:erenyiDiv}
  \leanok
  \lean{ProbabilityTheory.erenyiDiv_anti}
Let $\mathcal{P}' \subseteq \mathcal{P}$ and $\mathcal{Q}' \subseteq \mathcal{Q}$.
Then, for all $\alpha \in (0, 1)$,
\begin{align*}
  \eR_\alpha(\mathcal{P}', \mathcal{Q}')
  &\ge \eR_\alpha(\mathcal{P}, \mathcal{Q})
  \: .
\end{align*}
\end{lemma}
