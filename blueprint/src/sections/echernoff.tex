\chapter{e-Rényi and e-Chernoff divergences}

\begin{definition}\label{def:erenyiDiv}
  \uses{def:maxUtility, def:logUtility}
  \leanok
  \lean{ProbabilityTheory.erenyiDiv}
The e-Rényi divergence of order $\alpha \in (0,1)$ between two sets of probability measures $\mathcal{P}, \mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ is defined as
\begin{align*}
  \eR_\alpha(\mathcal{P}, \mathcal{Q})
  &= \frac{1}{1 - \alpha} \inf_{R \in \mathcal{P}(\mathcal{X})} \left( \alpha \mathcal{U}(R, \mathcal{P}) + (1 - \alpha) \mathcal{U}(R, \mathcal{Q}) \right)
  \: ,
\end{align*}
for $\mathcal{U}_{\log}$ the max-utility corresponding to the logarithmic utility.
\end{definition}


\begin{definition}\label{def:echernoffDiv}
  \uses{def:maxUtility, def:logUtility}
  \leanok
  \lean{ProbabilityTheory.echernoffDiv}
\begin{align*}
  \eC_\alpha(\mathcal{P}, \mathcal{Q})
  &= \inf_{R \in \mathcal{P}(\mathcal{X})} \max\left\{\mathcal{U}(R, \mathcal{P}), \mathcal{U}_{\log}(R, \mathcal{Q}) \right\}
  \: .
\end{align*}
\end{definition}

\begin{lemma}\label{lem:erenyiDiv_eq_iInf_acSet}
  \uses{def:erenyiDiv, def:acSet}
For $\alpha \in (0,1)$, if $\eR_\alpha(\mathcal{P}, \mathcal{Q}) < \infty$, then
\begin{align*}
  \eR_\alpha(\mathcal{P}, \mathcal{Q})
  &= \frac{1}{1 - \alpha} \inf_{R \in \mathcal{P}(\mathcal{X}), \: R \ll \mathcal{P}, \: R \ll \mathcal{Q}} \left( \alpha \mathcal{U}_{\log}(R, \mathcal{P}) + (1 - \alpha) \mathcal{U}_{\log}(R, \mathcal{Q}) \right)
  \: .
\end{align*}
That is, the infimum defining $\eR_\alpha(\mathcal{P}, \mathcal{Q})$ can be restricted to measures $R$ that are absolutely continuous with respect to both $\mathcal{P}$ and $\mathcal{Q}$.

TODO: can we drop the finiteness assumption?
\end{lemma}

\begin{proof}
  \uses{lem:maxUtility_eq_top_of_not_acSet}
If $R$ is not absolutely continuous with respect to $\mathcal{P}$, then $\mathcal{U}_{\log}(R, \mathcal{P}) = \infty$ by Lemma~\ref{lem:maxUtility_eq_top_of_not_acSet}.
Similarly, if $R$ is not absolutely continuous with respect to $\mathcal{Q}$, then $\mathcal{U}_{\log}(R, \mathcal{Q}) = \infty$.
Hence, in both cases, the term $\alpha \mathcal{U_{\log}(R, \mathcal{P}) + (1 - \alpha) \mathcal{U}_{\log}(R, \mathcal{Q})}$ is infinite and does not contribute to the infimum defining $\eR_\alpha(\mathcal{P}, \mathcal{Q})$.
\end{proof}


\begin{corollary}\label{cor:erenyiDiv_eq_iInf_support}
  \uses{def:erenyiDiv}
For $\alpha \in (0,1)$, if $\eR_\alpha(\mathcal{P}, \mathcal{Q}) < \infty$ and there exists a measurable set $A$ with $P(A^c) = 0$ and $Q(A^c) = 0$ for all $P \in \mathcal{P}$ and $Q \in \mathcal{Q}$, then
\begin{align*}
  \eR_\alpha(\mathcal{P}, \mathcal{Q})
  &= \frac{1}{1 - \alpha} \inf_{R \in \mathcal{P}(\mathcal{X}), \: R(A^c) = 0} \left( \alpha \mathcal{U}_{\log}(R, \mathcal{P}) + (1 - \alpha) \mathcal{U}_{\log}(R, \mathcal{Q}) \right)
  \: .
\end{align*}
That is, the infimum defining $\eR_\alpha(\mathcal{P}, \mathcal{Q})$ can be restricted to measures $R$ supported on $A$.
\end{corollary}

\begin{proof}
  \uses{lem:erenyiDiv_eq_iInf_acSet}
We apply Lemma~\ref{lem:erenyiDiv_eq_iInf_acSet} and note that any measure $R$ that is absolutely continuous with respect to both $\mathcal{P}$ and $\mathcal{Q}$ must satisfy $R(A^c) = 0$.
\end{proof}


\begin{lemma}\label{lem:erenyiDiv_map_eq}
  \uses{def:erenyiDiv}
Let $\phi : \mathcal{X} \to \mathcal{Y}$ be a measurable embedding. Then
\begin{align*}
  \eR_\alpha(\phi_*\mathcal{P}, \phi_*\mathcal{Q})
  &= \eR_\alpha(\mathcal{P}, \mathcal{Q})
  \: .
\end{align*}
\end{lemma}

\begin{proof}
  \uses{lem:erenyiDiv_map_le, cor:erenyiDiv_eq_iInf_support, lem:maxUtility_map_eq}
By Lemma~\ref{lem:erenyiDiv_map_le}, we have $\eR_\alpha(\phi_*\mathcal{P}, \phi_*\mathcal{Q}) \le \eR_\alpha(\mathcal{P}, \mathcal{Q})$.
Then either both sides are infinite and we are done, or the left hand side is finite.

Let $A = \phi(\mathcal{X}) \subseteq \mathcal{Y}$.
Then for all $P \in \mathcal{P}$ and $Q \in \mathcal{Q}$, we have $\phi_* P(A^c) = 0$ and $\phi_* Q(A^c) = 0$.
By Corollary~\ref{cor:erenyiDiv_eq_iInf_support}, we can restrict the infimum defining $\eR_\alpha(\phi_*\mathcal{P}, \phi_*\mathcal{Q})$ to measures supported on $A$.
Since $\phi$ is an embedding, every measure supported on $A$ is of the form $\phi_* R$ for some $R \in \mathcal{P}(\mathcal{X})$ (take $R = \phi^{-1}_* R'$, in which $\phi^{-1}$ is a partial inverse on the range of $\phi$).
\begin{align*}
  \eR_\alpha(\phi_*\mathcal{P}, \phi_*\mathcal{Q})
  &= \frac{1}{1 - \alpha} \inf_{R' \in \mathcal{P}(\mathcal{Y}), \: R'(A^c) = 0} \left( \alpha \mathcal{U}_{\log}(R', \phi_* \mathcal{P}) + (1 - \alpha) \mathcal{U}_{\log}(R', \phi_* \mathcal{Q}) \right)
  \\
  &= \frac{1}{1 - \alpha} \inf_{R \in \mathcal{P}(\mathcal{X})} \left( \alpha \mathcal{U}_{\log}(\phi_* R, \phi_* \mathcal{P}) + (1 - \alpha) \mathcal{U}_{\log}(\phi_* R, \phi_* \mathcal{Q}) \right)
  \: .
\end{align*}
Using then Lemma~\ref{lem:maxUtility_map_eq}, we get
\begin{align*}
  \eR_\alpha(\phi_*\mathcal{P}, \phi_*\mathcal{Q})
  &= \frac{1}{1 - \alpha} \inf_{R \in \mathcal{P}(\mathcal{X})} \left( \alpha \mathcal{U}_{\log}(R, \mathcal{P}) + (1 - \alpha) \mathcal{U}_{\log}(R, \mathcal{Q}) \right)
  \\
  &= \eR_\alpha(\mathcal{P}, \mathcal{Q})
  \: .
\end{align*}
\end{proof}

\begin{lemma}\label{lem:erenyiDiv_map_swap}
  \uses{def:erenyiDiv}
  \leanok
  \lean{ProbabilityTheory.erenyiDiv_of_involutive}
Let $\mathcal{P},\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ be two sets of probability measures.
Suppose that there exists a measurable map $\phi : \mathcal{X} \to \mathcal{X}$ such that $\phi \circ \phi = \id$ and $\phi_* \mathcal{P} = \mathcal{Q}$.
Then
\begin{align*}
  \eR_{1/2}(\mathcal{P}, \mathcal{Q})
  &= 2 \inf_{R \in \mathcal{P}(\mathcal{X}), \: \phi_* R = R} \mathcal{U}_{\log}(R, \mathcal{P})
  \: .
\end{align*}
\end{lemma}

\begin{proof}\leanok
  \uses{lem:maxUtility_map_le, lem:convex_maxUtility, lem:maxUtility_map_swap}
First consider an arbitrary $R \in \mathcal{P}(\mathcal{X})$.
Then $R' = \frac{1}{2}(R + \phi_* R)$ satisfies $\phi_* R' = R'$ and by convexity (Lemma~\ref{lem:convex_maxUtility}) and Lemma~\ref{lem:maxUtility_map_swap},
\begin{align*}
  \frac{1}{2} \mathcal{U}_{\log}(R', \mathcal{P}) + \frac{1}{2} \mathcal{U}_{\log}(R', \mathcal{Q})
  &\le \frac{1}{2} \left( \frac{1}{2} \mathcal{U}_{\log}(R, \mathcal{P}) + \frac{1}{2} \mathcal{U}_{\log}(R, \mathcal{Q}) \right)
  \\
  &\quad + \frac{1}{2} \left( \frac{1}{2} \mathcal{U}_{\log}(\phi_* R, \mathcal{P}) + \frac{1}{2} \mathcal{U}_{\log}(\phi_* R, \mathcal{Q}) \right)
  \\
  &= \frac{1}{2} \mathcal{U}_{\log}(R, \mathcal{P}) + \frac{1}{2} \mathcal{U}_{\log}(R, \mathcal{Q})
  \: .
\end{align*}
Hence we can restrict the infimum defining $\eR_{1/2}(\mathcal{P}, \mathcal{Q})$ to measures $R$ such that $\phi_* R = R$.
For such measures, by Lemma~\ref{lem:maxUtility_map_swap}, $\mathcal{U}_{\log}(R, \mathcal{Q}) = \mathcal{U}_{\log}(\phi_* R, \mathcal{P}) = \mathcal{U}_{\log}(R, \mathcal{P})$ and we get the result.
\end{proof}
