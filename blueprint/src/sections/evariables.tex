\paragraph{Notation}
Let $\mathcal{X}, \mathcal{Y}$ be measurable spaces.
We denote by $\mathcal{P}(\mathcal{X})$ the set of probability measures on $\mathcal{X}$ and write $\kappa : \mathcal{X} \rightsquigarrow \mathcal{Y}$ to mean that $\kappa$ is a Markov kernel from $\mathcal{X}$ to $\mathcal{Y}$.
$\mathbb{R}_{+,\infty}$ is the set of non-negative reals extended with $+\infty$ and $\overline{\mathbb{R}}$ is the set of reals extended with $-\infty$ and $+\infty$.

The central object of this study are E-variables, which are measurable functions that are bounded in expectation by $1$ for all probability measures in a set $\mathcal{Q}$.

\begin{definition}\label{def:eVar}
  \leanok
  \lean{MeasureTheory.IsEVar}
An E-variable for a set of probability measures $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ is a measurable function $f : \mathcal{X} \to \mathbb{R}_{+,\infty}$ such that for all $Q \in \mathcal{Q}$, $Q[f] \le 1$.
\end{definition}

We denote by $\mathcal{E}_{\mathcal{Q}}$ the set of E-variables for a set of probability measures $\mathcal{Q}$.

\begin{lemma}\label{lem:eVarSet_anti}
  \uses{def:eVar}
If $\mathcal{P} \subseteq \mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ then $\mathcal{E}_{\mathcal{Q}} \subseteq \mathcal{E}_{\mathcal{P}}$.
\end{lemma}

\end{proof}

We introduce randomized E-variables, which are Markov kernels that can be seen as randomized versions of E-variables.
The main difference is that they are allowed to be randomized, i.e., they can return a distribution over $\mathbb{R}_{+,\infty}$ instead of a single value.
This generalization is mainly useful thanks to the compositional properties it provides, as we will see in Lemma~\ref{lem:randEVar_comp}.

\begin{definition}\label{def:randEVar}
A randomized E-variable for a set of distributions $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ is a Markov kernel $\eta : \mathcal{X} \rightsquigarrow \mathbb{R}_{+,\infty}$ such that for all $Q \in \mathcal{Q}$, $(\eta \circ Q)[\id] \le 1$.
\end{definition}

We denote by $\mathcal{E}^R_{\mathcal{Q}}$ the set of randomized E-variables for a set of probability measures $\mathcal{Q}$.
Since measurable functions can be seen as deterministic Markov kernels, we have $\mathcal{E}_{\mathcal{Q}} \subseteq \mathcal{E}^R_{\mathcal{Q}}$.
Indeed, $Q[f] = (\delta_f \circ Q)[\id]$, in which $\delta_f : \mathcal{X} \rightsquigarrow \mathbb{R}_{+,\infty}$ is such that $\delta_f(x)$ is the dirac distribution at $f(x)$.


\begin{lemma}\label{lem:randEVar_iff_eVar}
  \uses{def:eVar, def:randEVar}
A Markov kernel $\eta : \mathcal{X} \rightsquigarrow \mathbb{R}_{+,\infty}$ is a randomized E-variable for a set of probability measures $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ if and only if the measurable function $f : \mathcal{X} \to \mathbb{R}_{+,\infty}$ defined by $f(x) = \eta(x)[\id]$ is an E-variable for $\mathcal{Q}$.
\end{lemma}

The next lemma shows that the larger set of randomized E-variables does not allow to achieve a larger expected utility than the set of E-variables.

\begin{lemma}\label{lem:sup_randEVar}
  \uses{def:eVar, def:randEVar}
Let $U : \mathbb{R}_{+,\infty} \to \overline{\mathbb{R}}$ be a concave function and $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ a set of probability measures.
Let $P \in \mathcal{P}(\mathcal{X})$. Then
\begin{align*}
  \sup_{\eta \in \mathcal{E}^R_{\mathcal{Q}}} (\eta \circ P)[U]
  &= \sup_{f \in \mathcal{E}_{\mathcal{Q}}} P[U \circ f]
  \: .
\end{align*}
\end{lemma}

\begin{proof}
Since $\mathcal{E}_{\mathcal{Q}} \subseteq \mathcal{E}^R_{\mathcal{Q}}$, the left hand side is greater than or equal to the right hand side.

For the other direction, let $\eta \in \mathcal{E}^R_{\mathcal{Q}}$ be a randomized E-variable.
By concavity of $U$,
\begin{align*}
  (\eta \circ P)[U]
  &= P[x \mapsto \eta(x)[U]]
  \le P[x \mapsto U(\eta(x)[\id])]
\end{align*}
The function $x \mapsto \eta(x)[\id]$ is a measurable function and belongs to $\mathcal{E}_{\mathcal{Q}}$.
\end{proof}



\section{Numeraire and duality}

This section gathers results from \cite{larsson2024numeraire}.


\begin{definition}\label{def:numeraire}
  \uses{def:eVar}
Let $P \in \mathcal{P}(\mathcal{X})$ and $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$.
A numeraire for $(P, \mathcal{Q})$ is a $P$-almost surely positive e-variable $f^* \in \mathcal{E}_{\mathcal{Q}}$ such that $P[f / f^*] \le 1$ for all $f \in \mathcal{E}_{\mathcal{Q}}$~.
\end{definition}


\begin{lemma}\label{lem:numeraire_iff_logOptimal}
  \uses{def:numeraire}
Let $f$ be a $P$-almost surely positive E-variable for a set of probability measures $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$.
Then $f$ is a numeraire for $(P, \mathcal{Q})$ if and only if $P[\log \frac{g}{f}] \le 0$ for all $g \in \mathcal{E}_{\mathcal{Q}}$.
\end{lemma}

\end{proof}


\begin{theorem}\label{thm:exists_numeraire}
  \uses{def:numeraire}
Let $P \in \mathcal{P}(\mathcal{X})$ and $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$.
A numeraire for $(P, \mathcal{Q})$ exists and is $P$-a.e. unique.
\end{theorem}

We will write $E^*_{P, \mathcal{Q}}$ for the numeraire for $(P, \mathcal{Q})$.


\begin{definition}\label{def:ripr}
  \uses{def:numeraire, thm:exists_numeraire}
Let $P \in \mathcal{P}(\mathcal{X})$ and $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$.
The reverse information projection (RIPr) of $P$ onto $\mathcal{Q}$ is the measure $P^*_{\mathcal{Q}}$ absolutely continuous with respect to $P$ such that
\begin{align*}
  \frac{dP^*_{\mathcal{Q}}}{dP} = \frac{1}{E^*_{P, \mathcal{Q}}} \: .
\end{align*}
\end{definition}


\begin{definition}\label{def:effectiveSet}
  \uses{def:eVar}
For a set of probability measures $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$, the effective set of measures $\mathcal{Q}_{\eff}$ is the set $\{\mu \in \mathcal{M}_+(\mathcal{X}) \mid \forall f \in \mathcal{E}_{\mathcal{Q}}, \: \mu[f] \le 1\}$.
\end{definition}

We could also define the effective set of measures for randomized E-variables, but it actually gives the same set.
