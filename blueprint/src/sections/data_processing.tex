
\begin{lemma}\label{lem:eVar_comp}
  \uses{def:eVar}
  \leanok
  \lean{ProbabilityTheory.IsEVar.comp}
Let $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ be a set of probability measures and let $\phi : \mathcal{X} \to \mathcal{Y}$ be a measurable function.
Then for all $g \in \mathcal{E}_{\phi_* \mathcal{Q}}$, $g \circ \phi \in \mathcal{E}_{\mathcal{Q}}$.
\end{lemma}

\begin{proof}\leanok
For $Q \in \mathcal{Q}$,
\begin{align*}
  Q[g \circ \phi]
  &= (\phi_* Q)[g]
  \le 1
  \: .
\end{align*}
\end{proof}


\begin{lemma}\label{lem:randEVar_comp}
  \uses{def:randEVar}
  \leanok
  \lean{ProbabilityTheory.IsRandEVar.comp}
Let $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ be a set of probability measures and let $\kappa : \mathcal{X} \rightsquigarrow \mathcal{Y}$ be a Markov kernel.
Then for all $\xi \in \mathcal{E}_{\kappa \circ \mathcal{Q}}^R$, $\xi \circ \kappa \in \mathcal{E}^R_{\mathcal{Q}}$.
\end{lemma}

\begin{proof}\leanok
For $Q \in \mathcal{Q}$,
\begin{align*}
  ((\xi \circ \kappa) \circ Q)[\id]
  &= (\xi \circ (\kappa \circ Q))[\id]
  \le 1
  \: .
\end{align*}
\end{proof}


\begin{theorem}[Data processing inequality with randomized e-variables]\label{thm:maxRandUtility_comp_le}
  \uses{def:maxRandUtility}
  \leanok
  \lean{ProbabilityTheory.maxRandUtility_comp_le}
Let $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ be a set of probability measures and $P \in \mathcal{P}(\mathcal{X})$.
Let $U : \mathbb{R}_{+,\infty} \to \overline{\mathbb{R}}$ be a utility function.
Then for all Markov kernels $\kappa : \mathcal{X} \rightsquigarrow \mathcal{Y}$,
\begin{align*}
  \mathcal{U}^R(P, \mathcal{Q})
  &\ge \mathcal{U}^R(\kappa \circ P, \kappa \circ \mathcal{Q})
  \: .
\end{align*}
\end{theorem}

\begin{proof}\leanok
  \uses{lem:randEVar_comp}
By Lemma~\ref{lem:randEVar_comp}, $\{\xi \circ \kappa \mid \xi \in \mathcal{E}^R_{\kappa \circ \mathcal{Q}}\} \subseteq \mathcal{E}^R_{\mathcal{Q}}$, hence
\begin{align*}
  \sup_{\eta \in \mathcal{E}^R_{\mathcal{Q}}} (\eta \circ P)[U]
  &\ge \sup_{\eta \in \{\xi \circ \kappa \mid \xi \in \mathcal{E}^R_{\kappa \circ \mathcal{Q}}\}} (\eta \circ P)[U]
  = \sup_{\xi \in \mathcal{E}^R_{\kappa \circ \mathcal{Q}}} (\xi \circ \kappa \circ P)[U]
  \: .
\end{align*}
\end{proof}


\begin{lemma}[Data processing inequality with e-variables]\label{lem:maxUtility_comp_le}
  \uses{def:maxUtility}
  \leanok
  \lean{ProbabilityTheory.maxUtility_comp_le}
Let $\kappa : \mathcal{X} \rightsquigarrow \mathcal{Y}$ be a Markov kernel, $P \in \mathcal{P}(\mathcal{X})$ and $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$.
Then
\begin{align*}
  \mathcal{U}(\kappa \circ P, \kappa \circ \mathcal{Q})
  &\le \mathcal{U}(P, \mathcal{Q})
  \: .
\end{align*}
\end{lemma}

\begin{proof}\leanok
  \uses{lem:maxRandUtility_eq_maxUtility, thm:maxRandUtility_comp_le}
We can use Lemma~\ref{lem:maxRandUtility_eq_maxUtility} to obtain the data-processing with a Markov kernel for e-variables from the result for randomized e-variables (Theorem~\ref{thm:maxRandUtility_comp_le}).
\end{proof}

  \uses{def:maxUtility}
  \leanok
  \lean{ProbabilityTheory.maxUtility_map_le}
Let $f : \mathcal{X} \to \mathcal{Y}$ be a measurable function, $P \in \mathcal{P}(\mathcal{X})$ and $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$.
Then
\begin{align*}
  \mathcal{U}(f_*P, f_* \mathcal{Q})
  &\le \mathcal{U}(P, \mathcal{Q})
  \: .
\end{align*}
\end{lemma}

\begin{proof}\leanok
  \uses{lem:maxUtility_comp_le}
Use Lemma~\ref{lem:maxUtility_comp_le} with the deterministic Markov kernel given by $f$.
\end{proof}

\begin{lemma}\label{lem:maxUtility_map_swap}
  \uses{def:maxUtility}
  \leanok
  \lean{ProbabilityTheory.maxUtility_involutive}
Let $R \in \mathcal{P}(\mathcal{X})$, $\mathcal{P} \subseteq \mathcal{P}(\mathcal{X})$ and let $\phi : \mathcal{X} \to \mathcal{X}$ be a measurable map such that $\phi \circ \phi = \id$.
Then
\begin{align*}
  \mathcal{U}(R, \phi_* \mathcal{P})
  &= \mathcal{U}(\phi_* R, \mathcal{P})
  \: .
\end{align*}
\end{lemma}

\begin{proof}\leanok
  \uses{lem:maxUtility_map_le}
To prove the first inequality, we use Lemma~\ref{lem:maxUtility_map_le}:
\begin{align*}
  \mathcal{U}(R, \phi_* \mathcal{P})
  &= \mathcal{U}(\phi_* (\phi_* R), \phi_* \mathcal{P})
  \\
  &\le \mathcal{U}(\phi_* R, \mathcal{P})
  \: .
\end{align*}
For the reverse inequality,
\begin{align*}
  \mathcal{U}(\phi_* R, \mathcal{P})
  &= \mathcal{U}(\phi_* R, \phi_* (\phi_* \mathcal{P}))
  \\
  &\le \mathcal{U}(R, \phi_* \mathcal{P})
  \: .
\end{align*}
\end{proof}
