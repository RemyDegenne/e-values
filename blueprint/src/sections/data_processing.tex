\section{Maximal logarithmic utility}
\label{sec:maximal_logarithmic_utility}


\begin{definition}\label{def:maxUtility}
The maximal utility of a probability measure $P \in \mathcal{P}(\mathcal{X})$ with respect to a set of probability measures $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ for a function $U : \mathbb{R}_{+,\infty} \to \overline{\mathbb{R}}$ is defined as
\begin{align*}
  \mathcal{U}(P, \mathcal{Q})
  &= \sup_{f \in \mathcal{E}_{\mathcal{Q}}} P[U \circ f]
  \: .
\end{align*}
Often, $U = \log$ will be used, in which case we talk about the maximal logarithmic utility.
\end{definition}

\begin{lemma}\label{lem:minimax_limsup_le}
  \uses{def:maxUtility}
Let $U : \mathbb{R}_{+,\infty} \to \overline{\mathbb{R}}$ be a continuous, increasing, concave function finite on $(0, \infty)$ and such that for $K>0$, $\int_{K}^\infty \frac{1}{U^{-1}(x)} dx < \infty$ and tends to 0 as $K \to \infty$ (for example $U = \log$).
Let $\mathcal{G}$ be a set of measurable functions from $\mathcal{X}$ to $\mathbb{R}_{+, \infty}$ and
let $P$ be a probability distribution on $\mathcal{X}$ such that $\sup_{f \in \mathcal{G}}P[f]$ is finite.
Then for all sequences $(f_n)_{n \in \mathbb{N}}$ of functions in $\mathcal{G}$, we have
\begin{align*}
  \limsup_{n \to \infty} P[U \circ f_n]
  &\le P[\limsup_{n \to \infty} U \circ f_n]
  \: .
\end{align*}
\end{lemma}

\begin{proof}
We check that the sequence of negative parts of the functions $-U \circ f_n$ is uniformly integrable in order to apply Fatou's lemma \cite{feinberg2020fatou}.
For $f \in \mathcal{G}$, that negative part is $(-U \circ f)^{-} = \max\{0, U \circ f\}$. For $K \ge 0$,
\begin{align*}
  P[\max\{0, U \circ f\} \mathbb{I}_{\max\{0, U \circ f\} \ge K}]
  &= P[ U \circ f \mathbb{I}_{U \circ f \ge K}]
  \\
  &= \int_{K}^\infty P\{U(f(x)) \ge x\} dx
  \\
  &= \int_{K}^\infty P\{f(x) \ge U^{-1}(x)\} dx
  \\
  &\le \int_{K}^\infty \frac{1}{U^{-1}(x)} P[f] dx
  \\
  &= \left(\int_{K}^\infty \frac{1}{U^{-1}(x)} dx\right) P[f]
  \: ,
\end{align*}
in which the inequality is due to Markov's inequality.
Then since $(P[f_n])_{n \in \mathbb{N}}$ is bounded from above (and below since $f \ge 0$) and $\int_{K}^\infty \frac{1}{U^{-1}(x)} dx \to_{K \to \infty} 0$, we have uniform integrability.
\end{proof}


Remark on the topology of pointwise convergence: for $X$ a set and $Y$ a Hausdorff space, the set of functions from $X$ to $Y$ is Hausdorff for the topology of pointwise convergence.
A set $A$ of functions is compact in that topology if and only if it is closed and each set $A_x = \{f(x) \mid f \in A\}$ is compact.


\begin{theorem}\label{thm:sion_maxUtility}
Let $U : \mathbb{R}_{+,\infty} \to \overline{\mathbb{R}}$ be a continuous, increasing, concave function finite on $(0, \infty)$ and such that for $K>0$, $\int_{K}^\infty \frac{1}{U^{-1}(x)} dx < \infty$ and tends to 0 as $K \to \infty$ (for example $U = \log$).
Let $\mathcal{G}$ be a convex set of measurable functions from $\mathcal{X}$ to $\mathbb{R}_{+, \infty}$, which is compact in the topology of pointwise convergence.
Let $\mathcal{P}$ be a convex set of probability distributions on $\mathcal{X}$ such that for all $P \in \mathcal{P}$, $\sup_{f \in \mathcal{G}}P[f]$ is finite.
Then
\begin{align*}
  \inf_{P \in \mathcal{P}} \sup_{f \in \mathcal{G}} P[U \circ f]
  &= \sup_{f \in \mathcal{G}} \inf_{P \in \mathcal{P}} P[U \circ f]
  \: .
\end{align*}
\end{theorem}

\begin{proof}
  \uses{thm:minimax, lem:minimax_limsup_le}
This is an application of the minimax theorem~\ref{thm:minimax} to the function $-P[U \circ f]$, for which we need to check the hypotheses.
$P, f \mapsto P[U \circ f]$ is linear in $P$ (hence convex) and concave in $f$ by concavity of $U$.
It only remains to check that $f \mapsto P[U \circ f]$ is upper semi-continuous.
By Lemma~\ref{lem:minimax_limsup_le}, for a sequence $(f_n)$ of functions in $\mathcal{G}$ which converges pointwise to $f \in \mathcal{G}$ we have
\begin{align*}
  \limsup_{n \to \infty} P[U \circ f_n]
  &\le P[\limsup_{n \to \infty} U \circ f_n]
  = P[U \circ f]
  \: .
\end{align*}
\end{proof}


\begin{lemma}\label{lem:minimax_fin_gen}
Let $\Phi = \{\phi_1, \ldots, \phi_d\}$ be a set of measurable functions from $\mathcal{X}$ to $\mathbb{R}$.
Let $\Lambda = \{\lambda \in \mathbb{R}_+^d \mid \forall x, 1 + \sum_i \lambda_i \phi_i(x) \ge 0\}$ and assume that it is a compact set.
Let $\mathcal{R}$ be a convex set of probability measures on $\mathcal{X}$ such that for all $R \in \mathcal{R}$ and $i \in [d]$, $R[\vert \phi_i \vert] < \infty$.
Then
\begin{align*}
  \inf_{R \in \mathcal{R}} \sup_{\lambda \in \Lambda} R[\log (1 + \sum_i \lambda_i \phi_i)]
  &= \sup_{\lambda \in \Lambda} \inf_{R \in \mathcal{R}} R[\log (1 + \sum_i \lambda_i \phi_i)]
  \: .
\end{align*}
\end{lemma}

\begin{proof}
  \uses{thm:sion_maxUtility}
We use Theorem~\ref{thm:sion_maxUtility} with $U(x) = \log(x)$ and $\mathcal{G} = \{f \mid \exists \lambda \in \Lambda, f = 1 + \sum_i \lambda_i \phi_i\}$.
That set is convex and compact in the topology of pointwise convergence.
We need to check the condition on the integrals against measures in $\mathcal{R}$: for $R \in \mathcal{R}$ and $f = 1 + \sum_i \lambda_i \phi_i$,
\begin{align*}
  R[f]
  &= 1 + \sum_i \lambda_i R[\phi_i]
  \le 1 + \sup_{\lambda \in \Lambda} \Vert \lambda \Vert_\infty \sum_i R[\vert \phi_i \vert]
  < \infty
  \: .
\end{align*}

\end{proof}


\section{Data-processing inequality for e-variables}
\label{sec:data_processing_eVar}


\begin{lemma}\label{lem:randEVar_comp}
  \uses{def:randEVar}
  \leanok
  \lean{ProbabilityTheory.IsEVar.comp, ProbabilityTheory.IsRandEVar.comp}
Let $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ be a set of probability measures and let $\phi : \mathcal{X} \to \mathcal{Y}$ be a measurable function and $\kappa : \mathcal{X} \rightsquigarrow \mathcal{Y}$ be a Markov kernel.
Then
\begin{enumerate}
  \item for all $g \in \mathcal{E}_{\phi \circ \mathcal{Q}}$, $g \circ \phi \in \mathcal{E}_{\mathcal{Q}}$,
  \item for all $\xi \in \mathcal{E}_{\kappa \circ \mathcal{Q}}^R$, $\xi \circ \kappa \in \mathcal{E}^R_{\mathcal{Q}}$.
\end{enumerate}
\end{lemma}

\begin{proof}\leanok
For $Q \in \mathcal{Q}$,
\begin{align*}
  Q[g \circ \phi]
  &= (\phi \circ Q)[g]
  \le 1
  \: , \\
  ((\xi \circ \kappa) \circ Q)[\id]
  &= (\xi \circ (\kappa \circ Q))[\id]
  \le 1
  \: .
\end{align*}
\end{proof}


\begin{theorem}[Data processing inequality with e-variables]\label{thm:dataprocessing}
  \uses{def:eVar}
  \leanok
  \lean{ProbabilityTheory.maxUtility_map_le, ProbabilityTheory.maxRandUtility_comp_le,
    ProbabilityTheory.maxUtility_comp_le}
Let $\mathcal{Q} \subseteq \mathcal{P}(\mathcal{X})$ be a set of probability measures and $P \in \mathcal{P}(\mathcal{X})$.
Let $U : \mathbb{R}_{+,\infty} \to \overline{\mathbb{R}}$.
Then for all measurable functions $\phi : \mathcal{X} \to \mathcal{Y}$ and all Markov kernels $\kappa : \mathcal{X} \rightsquigarrow \mathcal{Y}$,
\begin{align*}
  \sup_{f \in \mathcal{E}_{\mathcal{Q}}} P[U \circ f]
  &\ge \sup_{g \in \mathcal{E}_{\phi \circ \mathcal{Q}}} (\phi \circ P)[U \circ g]
  \: , \\
  \sup_{\eta \in \mathcal{E}^R_{\mathcal{Q}}} (\eta \circ P)[U]
  &\ge \sup_{\xi \in \mathcal{E}^R_{\kappa \circ \mathcal{Q}}} (\xi \circ \kappa \circ P)[U]
  \: .
\end{align*}
Furthermore, if $U$ is concave, then for all Markov kernels $\kappa : \mathcal{X} \rightsquigarrow \mathcal{Y}$,
\begin{align*}
  \sup_{f \in \mathcal{E}_{\mathcal{Q}}} P[U \circ f]
  &\ge \sup_{g \in \mathcal{E}_{\kappa \circ \mathcal{Q}}} (\kappa \circ P)[U \circ g]
  \: .
\end{align*}
For $\phi$ a measurable embedding, the inequalities are equalities.
\end{theorem}

\begin{proof}
  \uses{lem:randEVar_comp, lem:sup_randEVar}
By Lemma~\ref{lem:randEVar_comp}, $\{\xi \circ \kappa \mid \xi \in \mathcal{E}^R_{\kappa \circ \mathcal{Q}}\} \subseteq \mathcal{E}^R_{\mathcal{Q}}$, hence
\begin{align*}
  \sup_{\eta \in \mathcal{E}^R_{\mathcal{Q}}} (\eta \circ P)[U]
  &\ge \sup_{\eta \in \{\xi \circ \kappa \mid \xi \in \mathcal{E}^R_{\kappa \circ \mathcal{Q}}\}} (\eta \circ P)[U]
  = \sup_{\xi \in \mathcal{E}^R_{\kappa \circ \mathcal{Q}}} (\xi \circ \kappa \circ P)[U]
  \: .
\end{align*}
The argument for the first inequality is similar.

If $U$ is concave, we can use Lemma~\ref{lem:sup_randEVar} to obtain the data-processing with a Markov kernel for e-variables from the result for randomized e-variables.
\end{proof}

\paragraph{KL data processing from e-variables}

The data processing inequality for the numeraire utility implies the data processing inequality for KL divergences.

\begin{theorem}\label{thm:dataprocessing_kl}
  \uses{def:eVar}
  Let $P, Q \in \mathcal{P}(\mathcal{X})$.
  Then for all Markov kernels $\kappa : \mathcal{X} \rightsquigarrow \mathcal{Y}$,
  \begin{align*}
    \KL(P, Q)
    &\ge \KL(\kappa \circ P, \kappa \circ Q)
    \: .
  \end{align*}
\end{theorem}

\begin{proof}
  \uses{lem:sup_eVar_singleton, thm:dataprocessing}
By Lemma~\ref{lem:sup_eVar_singleton}, $\KL(P, Q) = \sup_{f \in \mathcal{E}_{\{Q\}}} P[\log f]$.
Then by Theorem~\ref{thm:dataprocessing}, we have
\begin{align*}
  \sup_{f \in \mathcal{E}_{\{Q\}}} P[\log f]
  &\ge \sup_{g \in \mathcal{E}_{\{\kappa \circ Q\}}} (\kappa \circ P)[\log g]
  \: .
\end{align*}
And that last term is $\KL(\kappa \circ P, \kappa \circ Q)$, again by Lemma~\ref{lem:sup_eVar_singleton}.
\end{proof}
